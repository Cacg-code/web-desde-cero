Ejercicio: Consumir APIs con fetch y variables de entorno

  • ⏱ 60 minutos
  • 📶 Nivel: intermedio

Objetivo

Consultar una API real desde un script de Node, guardar la configuración en un .env y escribir en JavaScript puro las utilidades que sostienen ese trabajo.

Cómo se trabaja

La parte A es práctica real en tu terminal (necesita internet). La parte B son cuatro funciones de JavaScript puro: un lector de .env, un constructor de URLs, un obtenerJson al que se le pasa fetch como argumento (así se puede probar sin red) y un lector de configuración. Pega solo el JavaScript, sin <script>.

    function parsearEnv(texto) {
  const resultado = {};
  for (const linea of texto.split("\n")) {
    const l = linea.trim();
    if (l === "" || l.startsWith("#")) continue;
    const i = l.indexOf("=");
    if (i === -1) continue;
    const clave = l.slice(0, i).trim();
    let valor = l.slice(i + 1).trim();
    if (valor.length >= 2 && valor.startsWith('"') && valor.endsWith('"')) {
      valor = valor.slice(1, -1);
    }
    resultado[clave] = valor;
  }
  return resultado;
}

function construirUrl(base, params) {
  const limpio = {};
  for (const [k, v] of Object.entries(params)) {
    if (v !== undefined && v !== null) limpio[k] = v;
  }
  const query = new URLSearchParams(limpio).toString();
  return query ? base + "?" + query : base;
}

async function obtenerJson(fetchFn, url) {
  const respuesta = await fetchFn(url);
  if (!respuesta.ok) {
    throw new Error("HTTP " + respuesta.status);
  }
  return respuesta.json();
}

function leerConfig(env) {
  return {
    puerto: Number(env.PORT) || 3000,
    modo: env.NODE_ENV ?? "development",
  };
}
